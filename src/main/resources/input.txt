PARSER_START
<stmt_list> ::= <stmt_list> '+' <stmt> | <stmt> ;
<stmt> ::= <assign> | <print> ;
<assign> ::= <identifier> '=' <expr> ';' ;
<print> ::= <PRINT> '(' <identifier> ')' ';' ;
<expr> ::= <term> {[('+' <term>)]} ;
<term> ::= <number> '+' | <identifier> ;
<number> ::= <NUMBER> ;
<identifier> ::= <IDENTIFIER> ;
PARSER_END

LEXER_START
<IDENTIFIER> ::= [a-zA-Z_][a-zA-Z0-9_]* ;
<NUMBER> ::= [0-9]+ ;
<PRINT> ::= print ;
SKIP_CHARS
<SKIPCHARS> ::= [\t\r\n]+;
LEXER_END
